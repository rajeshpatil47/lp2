#include <bits/stdc++.h>
using namespace std;

void findBFS(vector<vector<int>>&adj,int n){
    queue<int>q;
    int vis[n]={0};
    q.push(0);
    vis[0]=1;
    
    while(!q.empty()){
        int curnode = q.front();
        q.pop();
        cout<<curnode<<" ";
        
        for(auto it:adj[curnode]){
            if(!vis[it]){
                q.push(it);
                vis[it]=1;
            }
        }
    }
    return;
}

int main()
{
    int n,m;
    cout<<"Enter the number of nodes and edges : "<<endl;
    cin>>n>>m;
    
    vector<vector<int>>adj(n);
    
    for(int i=0;i<m;i++){
        int u,v;
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    for(int i=0;i<n;i++){
        cout<<i<<" -> ";
        for(auto it:adj[i])cout<<it<<" ";
        cout<<endl;
    }
    
    findBFS(adj,n);
    cout<<"This is BFS of the graph.";
    
    return 0;
}
