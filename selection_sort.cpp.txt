#include<bits/stdc++.h>
using namespace std;


void sort(vector<int>&v){
    int min=0;
    int n=v.size();
    for(int i=0;i<n;i++){
        int min=i;
        for(int j=i+1;j<n;j++){
            if(v[j]<v[min])min=j;
        }
        swap(v[min],v[i]);
        
    }
    return;
}

int main(){
    vector<int>v={12,1,5,22,4,55,100};
    sort(v);
    for(auto it:v){
        cout<<it<<" ";
    }
}
